```bash
  generate-mm)
    # Multimodal AI draft from tech-pack images/PDF (handoff v1.1 §4).
    # json fields → individual multipart form fields via node (same
    # no-eval-JSON-in-shell rationale as set-status); each --file → a
    # files=@ part. Server limits: <=8 files, <=12 MB each,
    # png/jpg/webp/gif/pdf; the <=8 is also enforced here.
    [ $# -ge 3 ] || { echo "usage: sam generate-mm <json> --file <path> [--file <path>]..." >&2; exit 2; }
    mmjson="$1"; shift
    mmfiles=()
    while [ $# -gt 0 ]; do
      case "$1" in
        --file)
          [ $# -ge 2 ] || { echo "--file needs a path" >&2; exit 2; }
          mmfiles+=("$2"); shift 2 ;;
        *) echo "unexpected argument '$1' (only --file <path> after the json)" >&2; exit 2 ;;
      esac
    done
    [ ${#mmfiles[@]} -ge 1 ] || { echo "at least one --file <path> is required" >&2; exit 2; }
    [ ${#mmfiles[@]} -le 8 ] || { echo "too many files: ${#mmfiles[@]} (server limit is 8)" >&2; exit 2; }
    for f in "${mmfiles[@]}"; do
      [ -f "$f" ] || { echo "file not found: $f" >&2; exit 2; }
    done
    # Validate the json up front — the emitter below runs in a process
    # substitution whose exit code set -e cannot see.
    node -e 'const o = JSON.parse(process.argv[1]); if (!o || typeof o !== "object" || Array.isArray(o)) { console.error("json must be an object"); process.exit(1); }' "$mmjson" \
      || { echo "invalid json for generate-mm" >&2; exit 2; }
    # Emit NUL-separated name=value pairs (nested objects JSON-encoded;
    # null/undefined skipped). --form-string, not -F, for these: a value
    # starting with @ or < must be sent literally, never read a local file.
    form_args=()
    while IFS= read -r -d '' pair; do
      form_args+=(--form-string "$pair")
    done < <(node -e '
      const o = JSON.parse(process.argv[1]);
      for (const [k, v] of Object.entries(o)) {
        if (v === null || v === undefined) continue;
        const s = typeof v === "object" ? JSON.stringify(v) : String(v);
        process.stdout.write(`${k}=${s}\u0000`);
      }' "$mmjson")
    for f in "${mmfiles[@]}"; do form_args+=(-F "files=@$f"); done
    # Opus multimodal draft: ~60-120 s, costs the SAM server's own credits.
    curl -sS --fail-with-body --max-time 180 -H "$AUTH" \
      -X POST "${form_args[@]}" "$BASE/analyses/generate-mm"
    ;;
```
